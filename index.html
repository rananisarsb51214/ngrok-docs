firestore.rules (Phase 3 Enforced Multi-Tenant RBAC)
Security Enforced 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Multi-tenant Org membership helper
    function isOrgMember(orgId) {
      return request.auth != null && 
        exists(/databases/$(database)/documents/orgs/$(orgId)/members/$(request.auth.uid));
    }

    function getOrgRole(orgId) {
      return get(/databases/$(database)/documents/orgs/$(orgId)/members/$(request.auth.uid)).data.role;
    }

    match /orgs/{orgId} {
      allow read: if isOrgMember(orgId);
      allow write: if isOrgMember(orgId) && getOrgRole(orgId) in ['owner', 'admin'];

      // Isolated Project subcollection
      match /projects/{projectId} {
        allow read: if isOrgMember(orgId);
        allow write: if isOrgMember(orgId) && getOrgRole(orgId) in ['owner', 'admin', 'member'];
      }

      // Usage Ledgers & Audit Logs (Append-only or Admin read)
      match /auditLogs/{logId} {
        allow read: if isOrgMember(orgId) && getOrgRole(orgId) in ['owner', 'admin'];
        allow write: if false; // Server-side writes only via Admin SDK
      }
    }
  }
}

Phase 7 Isolation Refinement

Decoupled MCP Architecture Boundary

Critical Architecture Decision: The Model Context Protocol (MCP) server is strictly decoupled and read-only. It is deliberately not placed inside or acting as the security perimeter of the core app.

Why Decoupling is Enforced:
  • Prevents Perimeter Collapse: If an MCP server is compromised or injected, internal DB & auth boundaries remain intact.
  • Read-Only Scoping: Exposes curated, safe tool schemas without executing arbitrary state mutations.
  • Independent Auditing: Separate telemetry stream for tool invocations and contextual reads.
